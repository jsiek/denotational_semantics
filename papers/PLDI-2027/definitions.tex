\section*{About this document}

This is a summary, for review, of the definitions in the Agda development
of the closure-conversion compiler in \texttt{agda/Compiler}.
It uses the notation of \texttt{talks/ITRS-2020} and named variables
instead of de Bruijn indices.

\paragraph{Contents.}
\begin{enumerate}
\item \textbf{Values and operators:}
  \begin{itemize}
  \item the value domain and environments, with $\rho_0(x) = \{\omega\}$;
  \item the operators $\star$, $\Lambda$, $\key{pair}$/$\key{car}$/$\key{cdr}$,
    strict $\mathcal{T}$, $\key{proj}$, $\mathcal{L}$/$\mathcal{R}$/$\mathcal{C}$,
    $\key{guard}$, and $\odot$;
  \item consistency ($\sim$, $\consis$) and continuity.
  \end{itemize}
\item \textbf{Languages:} the syntax of ISWIM and Clos1--Clos5, including Clos5 programs.
\item \textbf{Semantics:} the clauses all languages share, then each language's own
  clauses, including Clos5's table-indexed semantics, $\key{code}$, and $\key{table}$.
\item \textbf{Compilation:} annotate, enclose, optimize, concretize, delay (with the let),
  and globalize.
\item \textbf{Relations and lemma statements:}
  \begin{itemize}
  \item $\mathit{Enclosed}$, plus the annotate, enclose and optimize equalities;
  \item for concretize, the $\mathit{Ref}$ semantics and $\sim_\varphi$;
  \item for delay, $\mathit{depth}$/$\mathit{Bnd}$, the generic relation $\mathcal{R}_k$
    with its preservation and reflection instances, $\rhd$, and the let equation;
  \item for globalize, $\mathit{Extends}$, $\mathit{RB}_n$, and $\mathit{Agree}_n$;
  \item the end-to-end theorem.
  \end{itemize}
\end{enumerate}

\paragraph{Where it doesn't match the Agda literally.}
Using names instead of de Bruijn indices changes a few definitions:
\begin{itemize}
\item \textbf{Annotate} lists $\FV(\lam{y} N)$ as the closure's free variables.
  The Agda lists the first $m$ surrounding variables, enough to cover every
  variable the body uses.
\item \textbf{Closure free variables} are variables in Clos1 and Clos2 and
  references in Clos3, which is all the passes produce. The Agda allows
  arbitrary terms there.
\item \textbf{Optimize} has no renaming parameter. It keeps $x_i$ when
  $x_i \in \FV(N)$.
\item \textbf{Concretize}'s map has no fallback for variables past the free
  variables. Closed code can't refer to them.
\item \textbf{Tuple indices} run from $1$ to $n$.
\item \textbf{Global definitions} are listed oldest first, so $\funref{k}$ names $d_k$.
\item \textbf{Compiler correctness} is stated with $\obseq$.
  The Agda proves its two halves as separate theorems,
  \texttt{compile-correct-const} and \texttt{compile-correct-nonempty}.
\end{itemize}

\section{Values and Operators}

\subsection{Values}

\[
\begin{array}{rcl}
  c &\in& \text{constants} \qquad x, y, z, e, f \in \X \\[1ex]
  \V \ni d, u, v, w &::=& c \mid V \mapsto w \mid \nu \mid \omega
     \mid \carv{u} \mid \cdrv{v}
     \mid \tup{i}{v} \mid \Left{v} \mid \Right{v} \mid \emptytup \\
  U, V, W &\in& \FSET{\V} \\
  D, E, X, Y &\in& \SET{\V} \\
  F, G &:& \SET{\V} \to \SET{\V}
\end{array}
\]

\[
\begin{array}{rcl}
  \rho &:& \X \to \SET{\V} \\
  \rho_0(x) &=& \{\omega\} \\
  D \simeq E &\iff& D \subseteq E \text{ and } E \subseteq D \\
  \rho \simeq \rho' &\iff& \forall x.\; \rho(x) \simeq \rho'(x)
\end{array}
\]

\subsection{Operators}

\begin{align*}
  D \app E &= \{ w \mid \exists V \NE.\; (V \mapsto w) \in D,\; V \subseteq E \} \\
  \Lambda(F) &= \{ V \mapsto w \mid V \NE,\; w \in F(V) \} \cup \{ \nu \} \\
  \mathcal{B}(c) &= \{ c \} \\
  \key{pair}(D, E) &= \{ \carv{u} \mid u \in D,\; E \NE \}
    \cup \{ \cdrv{v} \mid D \NE,\; v \in E \} \\
  \key{car}(D) &= \{ u \mid \carv{u} \in D \} \\
  \key{cdr}(D) &= \{ v \mid \cdrv{v} \in D \} \\
  \mathcal{T}() &= \{ \emptytup \} \\
  \mathcal{T}(D_1,\ldots,D_n) &= \{ \tup{i}{d} \mid d \in D_i,\; \forall j.\; D_j \NE \}
     \qquad (n > 0) \\
  \key{proj}_i(D) &= \{ d \mid \tup{i}{d} \in D \} \\
  \mathcal{L}(D) &= \{ \Left{d} \mid d \in D \} \\
  \mathcal{R}(D) &= \{ \Right{d} \mid d \in D \} \\
  \mathcal{C}(D, F, G) &=
    \{ w \mid \exists V \NE.\; (\forall v \in V.\; \Left{v} \in D),\; w \in F(V) \} \\
    &\cup\; \{ w \mid \exists V \NE.\; (\forall v \in V.\; \Right{v} \in D),\; w \in G(V) \} \\
  \key{guard}(D_1,\ldots,D_n; E) &= \{ w \mid (\forall i.\; D_i \NE),\; w \in E \}
\end{align*}

\begin{align*}
  D \odot E &= (\key{car}(D) \app \key{cdr}(D)) \app E
\end{align*}

\subsection{Consistency}

\begin{align*}
  c \sim c' &\iff c = c' \\
  (V \mapsto w) \sim (V' \mapsto w') &\iff V \not\approx V' \;\lor\; (V \approx V' \land w \sim w') \\
  (V \mapsto w) \sim \nu, \quad \nu \sim (V \mapsto w), \quad \nu \sim \nu &\iff \mathsf{true} \\
  \omega \sim \omega, \quad \emptytup \sim \emptytup &\iff \mathsf{true} \\
  \carv{u} \sim \carv{v} &\iff u \sim v \\
  \carv{u} \sim \cdrv{v}, \quad \cdrv{v} \sim \carv{u} &\iff \mathsf{true} \\
  \cdrv{u} \sim \cdrv{v} &\iff u \sim v \\
  \tup{i}{d} \sim \tup{j}{d'} &\iff n = n' \land (i \neq j \lor d \sim d')
     \qquad (i \in \{1..n\},\; j \in \{1..n'\}) \\
  \Left{d} \sim \Left{d'} &\iff d \sim d' \\
  \Right{d} \sim \Right{d'} &\iff d \sim d' \\
  u \sim v &\iff \mathsf{false} \quad \text{otherwise} \\[1ex]
  V \approx V' &\iff \forall u \in V, v \in V'.\; u \sim v \\
  \consis(D) &\iff \forall u, v \in D.\; u \sim v
\end{align*}

\subsection{Continuity}

\begin{align*}
  \rho' \subseteq \rho &\iff \forall x.\; \rho'(x) \subseteq \rho(x) \\
  \mathit{finiteNE}(\rho') &\iff \forall x.\; \exists V \NE.\; \rho'(x) = V \\
  \mathit{continuous}(\mathcal{E}) &\iff
    \forall \rho\, v.\; (\forall x.\; \rho(x) \NE) \land v \in \mathcal{E}(\rho) \\
    &\qquad \implies \exists \rho'.\; \mathit{finiteNE}(\rho'),\; \rho' \subseteq \rho,\; v \in \mathcal{E}(\rho')
\end{align*}

\section{Languages}

\subsection{ISWIM}

The source language: a call-by-value $\lambda$-calculus with constants, tuples, and sums.
\begin{syntax}
  & L, M, N &::=& x \mid \lam{x} N \mid L\;M \mid c
    \mid \tuple{M_1,\ldots,M_n} \mid \prj{i} M \\
  &&& \mid \inl M \mid \inr M \mid \Case{L}{x}{M}{y}{N}
\end{syntax}
\[
  \FV(\lam{y} N) = \FV(N) \setminus \{y\}
\]

\subsection{Clos1}

Like ISWIM, but each $\lambda$ is annotated with its free variables $x_1\ldots x_n$; the body still refers to the surrounding variables directly.
\begin{syntax}
  & L, M, N &::=& x \mid \clos{x_1\ldots x_n}{y} N \mid L\;M \mid c
    \mid \tuple{M_1,\ldots,M_n} \mid \prj{i} M \\
  &&& \mid \inl M \mid \inr M \mid \Case{L}{x}{M}{y}{N}
\end{syntax}
\[
  \FV(\clos{x_1\ldots x_n}{y} N) = \{x_1, \ldots, x_n\} \cup (\FV(N) \setminus \{y\})
\]

\subsection{Clos2}

Each closure now binds the values of its free variables $z_1\ldots z_n$ to names $x_1\ldots x_n$, so its body refers only to those names and its parameter.
\begin{syntax}
  & L, M, N &::=& x \mid \clos{z_1\ldots z_n}{x_1\ldots x_n\, y} N \mid L\;M \mid c
    \mid \tuple{M_1,\ldots,M_n} \mid \prj{i} M \\
  &&& \mid \inl M \mid \inr M \mid \Case{L}{x}{M}{y}{N}
\end{syntax}
\[
  \FV(\clos{z_1\ldots z_n}{x_1\ldots x_n\, y} N) = \{z_1, \ldots, z_n\}
\]

\subsection{Clos3}

Each closure now packs references $r_1\ldots r_n$ to its free variables into one tuple bound to $e$, so its body reaches them by projections from $e$.
\begin{syntax}
  & r &::=& x \mid \prj{i} x \\
  & L, M, N &::=& x \mid \clos{r_1\ldots r_n}{e\, y} N \mid L\;M \mid c
    \mid \tuple{M_1,\ldots,M_n} \mid \prj{i} M \\
  &&& \mid \inl M \mid \inr M \mid \Case{L}{x}{M}{y}{N}
\end{syntax}
\[
  \FV(\clos{r_1\ldots r_n}{e\, y} N) = \FV(r_1) \cup \cdots \cup \FV(r_n)
\]

\subsection{Clos4}

Closures are explicit pairs of closed code $\fun{e\,y.\,N}$ and an environment, built with $\key{pair}$, taken apart with $\key{fst}$ and $\key{snd}$, and applied to an environment and an argument with $L(M, N)$; $\key{let}$ is used by the delayed application.
\begin{syntax}
  & L, M, N &::=& x \mid \fun{e\,y.\,N} \mid L(M, N) \mid c
    \mid \pair{M}{N} \mid \FST M \mid \SND M \\
  &&& \mid \tuple{M_1,\ldots,M_n} \mid \prj{i} M
    \mid \inl M \mid \inr M \mid \Case{L}{x}{M}{y}{N}
    \mid \Let x = M \In N
\end{syntax}
\[
  \FV(\fun{e\,y.\,N}) = \emptyset
\]

\subsection{Clos5}

Like Clos4, but the code is hoisted into a program's list of global definitions $d_k$, which terms refer to as $\funref{k}$.
\begin{syntax}
  & L, M, N &::=& x \mid \funref{k} \mid L(M, N) \mid c
    \mid \pair{M}{N} \mid \FST M \mid \SND M \\
  &&& \mid \tuple{M_1,\ldots,M_n} \mid \prj{i} M
    \mid \inl M \mid \inr M \mid \Case{L}{x}{M}{y}{N}
    \mid \Let x = M \In N \\[1ex]
  & d &::=& e\,y.\,N \qquad\qquad (\FV(e\,y.\,N) = \emptyset) \\
  & P &::=& (d_0, \ldots, d_{m-1};\; M)
\end{syntax}

\section{Semantics}

\begin{align*}
  \SEM{-}^{\mathrm{ISWIM}}, \SEM{-}^{\mathrm{1}}, \ldots, \SEM{-}^{\mathrm{4}}
    &: \mathit{Term} \to (\X \to \SET{\V}) \to \SET{\V} \\
  \SEM{-}^{\mathrm{5}}_T
    &: \mathit{Term} \to (\X \to \SET{\V}) \to \SET{\V}
    \qquad (T : \mathbb{N} \to \SET{\V}) \\
  \SEM{-}^{\mathrm{5}} &: \mathit{Program} \to \SET{\V}
\end{align*}
For each language $\ell \in \{\mathrm{ISWIM}, 1, 2, 3, 4\}$, $\SEM{-}^{\ell}$ is
defined by the common clauses below together with the clauses for $\ell$.

\subsection{Clauses common to all languages}

\begin{align*}
  \SEM{x}^{\ell}\rho &= \rho(x) \\
  \SEM{c}^{\ell}\rho &= \mathcal{B}(c) \\
  \SEM{\tuple{M_1,\ldots,M_n}}^{\ell}\rho &= \mathcal{T}(\SEM{M_1}^{\ell}\rho, \ldots, \SEM{M_n}^{\ell}\rho) \\
  \SEM{\prj{i} M}^{\ell}\rho &= \key{proj}_i(\SEM{M}^{\ell}\rho) \\
  \SEM{\inl M}^{\ell}\rho &= \mathcal{L}(\SEM{M}^{\ell}\rho) \\
  \SEM{\inr M}^{\ell}\rho &= \mathcal{R}(\SEM{M}^{\ell}\rho) \\
  \SEM{\Case{L}{x}{M}{y}{N}}^{\ell}\rho &=
     \mathcal{C}(\SEM{L}^{\ell}\rho,\; \lambda X.\, \SEM{M}^{\ell}\ext{\rho}{x}{X},\; \lambda Y.\, \SEM{N}^{\ell}\ext{\rho}{y}{Y})
\end{align*}

\subsection{ISWIM}

\begin{align*}
  \SEM{\lam{x} N}^{\mathrm{ISWIM}}\rho &= \Lambda(\lambda X.\, \SEM{N}^{\mathrm{ISWIM}}\ext{\rho}{x}{X}) \\
  \SEM{L\;M}^{\mathrm{ISWIM}}\rho &= \SEM{L}^{\mathrm{ISWIM}}\rho \app \SEM{M}^{\mathrm{ISWIM}}\rho
\end{align*}

\subsection{Clos1}

\begin{align*}
  \SEM{\clos{x_1\ldots x_n}{y} N}^{\mathrm{1}}\rho &=
    \key{guard}(\rho(x_1), \ldots, \rho(x_n);\;
       \Lambda(\lambda Y.\, \SEM{N}^{\mathrm{1}}\ext{\rho}{y}{Y})) \\
  \SEM{L\;M}^{\mathrm{1}}\rho &= \SEM{L}^{\mathrm{1}}\rho \app \SEM{M}^{\mathrm{1}}\rho
\end{align*}

\subsection{Clos2}

\begin{align*}
  \SEM{\clos{z_1\ldots z_n}{x_1\ldots x_n\, y} N}^{\mathrm{2}}\rho &=
    \key{guard}(D_1, \ldots, D_n;\;
       \Lambda(\lambda Y.\, \SEM{N}^{\mathrm{2}}\rho_0(x_1{:=}D_1,\ldots,x_n{:=}D_n, y{:=}Y)))
    \\ &\qquad\text{where } D_i = \rho(z_i) \\
  \SEM{L\;M}^{\mathrm{2}}\rho &= \SEM{L}^{\mathrm{2}}\rho \app \SEM{M}^{\mathrm{2}}\rho
\end{align*}

\subsection{Clos3}

\begin{align*}
  \SEM{\clos{r_1\ldots r_n}{e\, y} N}^{\mathrm{3}}\rho &=
    \Lambda(\lambda X.\, \Lambda(\lambda Y.\, \SEM{N}^{\mathrm{3}}\rho_0(e{:=}X, y{:=}Y)))
    \app \mathcal{T}(\SEM{r_1}^{\mathrm{3}}\rho, \ldots, \SEM{r_n}^{\mathrm{3}}\rho) \\
  \SEM{L\;M}^{\mathrm{3}}\rho &= \SEM{L}^{\mathrm{3}}\rho \app \SEM{M}^{\mathrm{3}}\rho
\end{align*}

\subsection{Clos4}

\begin{align*}
  \SEM{\fun{e\,y.\,N}}^{\mathrm{4}}\rho &= \Lambda(\lambda X.\, \Lambda(\lambda Y.\, \SEM{N}^{\mathrm{4}}\rho_0(e{:=}X, y{:=}Y))) \\
  \SEM{L(M, N)}^{\mathrm{4}}\rho &= (\SEM{L}^{\mathrm{4}}\rho \app \SEM{M}^{\mathrm{4}}\rho) \app \SEM{N}^{\mathrm{4}}\rho \\
  \SEM{\pair{M}{N}}^{\mathrm{4}}\rho &= \key{pair}(\SEM{M}^{\mathrm{4}}\rho, \SEM{N}^{\mathrm{4}}\rho) \\
  \SEM{\FST M}^{\mathrm{4}}\rho &= \key{car}(\SEM{M}^{\mathrm{4}}\rho) \\
  \SEM{\SND M}^{\mathrm{4}}\rho &= \key{cdr}(\SEM{M}^{\mathrm{4}}\rho) \\
  \SEM{\Let x = M \In N}^{\mathrm{4}}\rho &= \Lambda(\lambda X.\, \SEM{N}^{\mathrm{4}}\ext{\rho}{x}{X}) \app \SEM{M}^{\mathrm{4}}\rho
\end{align*}

\subsection{Clos5}

\begin{align*}
  \SEM{\funref{k}}^{\mathrm{5}}_T\rho &= T(k) \\
  \SEM{L(M, N)}^{\mathrm{5}}_T\rho &= (\SEM{L}^{\mathrm{5}}_T\rho \app \SEM{M}^{\mathrm{5}}_T\rho) \app \SEM{N}^{\mathrm{5}}_T\rho \\
  \SEM{\pair{M}{N}}^{\mathrm{5}}_T\rho &= \key{pair}(\SEM{M}^{\mathrm{5}}_T\rho, \SEM{N}^{\mathrm{5}}_T\rho) \\
  \SEM{\FST M}^{\mathrm{5}}_T\rho &= \key{car}(\SEM{M}^{\mathrm{5}}_T\rho) \\
  \SEM{\SND M}^{\mathrm{5}}_T\rho &= \key{cdr}(\SEM{M}^{\mathrm{5}}_T\rho) \\
  \SEM{\Let x = M \In N}^{\mathrm{5}}_T\rho &= \Lambda(\lambda X.\, \SEM{N}^{\mathrm{5}}_T\ext{\rho}{x}{X}) \app \SEM{M}^{\mathrm{5}}_T\rho
\end{align*}
together with the common clauses, with $\SEM{-}^{\mathrm{5}}_T$ in place of $\SEM{-}^{\ell}$.

\begin{align*}
  \key{code}_T(e\,y.\,N) &= \Lambda(\lambda X.\, \Lambda(\lambda Y.\, \SEM{N}^{\mathrm{5}}_T\rho_0(e{:=}X, y{:=}Y))) \\[1ex]
  \key{table}(d_0,\ldots,d_{m-1})(k) &=
    \begin{cases}
      \key{code}_{\key{table}(d_0,\ldots,d_{k-1})}(d_k) & k < m \\
      \emptyset & \text{otherwise}
    \end{cases} \\[1ex]
  \SEM{(d_0, \ldots, d_{m-1};\; M)}^{\mathrm{5}} &= \SEM{M}^{\mathrm{5}}_{\key{table}(d_0,\ldots,d_{m-1})}\,\rho_0
\end{align*}

\section{Compilation}

\[
  \pass{compile} = \pass{globalize} \circ \pass{delay} \circ \pass{concretize}_{\mathit{id}}
     \circ \pass{optimize} \circ \pass{enclose} \circ \pass{annotate}
\]

In each pass, the clauses not shown are homomorphic.

\subsection{Annotate (ISWIM $\to$ Clos1)}

\begin{align*}
  \pass{annotate}(\lam{y} N) &= \clos{x_1\ldots x_n}{y} \pass{annotate}(N)
    \qquad\text{where } x_1, \ldots, x_n = \FV(\lam{y} N)
\end{align*}

\subsection{Enclose (Clos1 $\to$ Clos2)}

\begin{align*}
  \pass{enclose}(\clos{x_1\ldots x_n}{y} N) &=
     \clos{x_1\ldots x_n}{x_1\ldots x_n\, y} \pass{enclose}(N)
\end{align*}

\subsection{Optimize (Clos2 $\to$ Clos2)}

\begin{align*}
  \pass{optimize}(\clos{z_1\ldots z_n}{x_1\ldots x_n\, y} N) &=
    \clos{z_{i_1}\ldots z_{i_k}}{x_{i_1}\ldots x_{i_k}\, y} \pass{optimize}(N) \\
  \text{where } \{ i_1 < \cdots < i_k \} &= \{ i \mid x_i \in \FV(N) \}
\end{align*}

\subsection{Concretize (Clos2 $\to$ Clos3)}

\[
  \varphi : \X \to \mathit{Ref}
  \qquad (\mathit{Ref} \text{ is the set of Clos3 references } r)
\]
\begin{align*}
  \pass{concretize}_\varphi(x) &= \varphi(x) \\
  \pass{concretize}_\varphi(\clos{z_1\ldots z_n}{x_1\ldots x_n\, y} N) &=
    \clos{\varphi(z_1)\ldots\varphi(z_n)}{e\,y}
       \pass{concretize}_{\varphi_n}(N) \\
  \text{where } \varphi_n &= \{ y \mapsto y,\; x_1 \mapsto \prj{1} e,\; \ldots,\; x_n \mapsto \prj{n} e \},
    \quad e \text{ fresh} \\
  \pass{concretize}_\varphi(\Case{L}{x}{M}{y}{N}) &=
    \key{case}\;\pass{concretize}_\varphi(L)\;\key{of} \\
    &\qquad \key{inl}\,x \Rightarrow \pass{concretize}_{\ext{\varphi}{x}{x}}(M)
     \mid \key{inr}\,y \Rightarrow \pass{concretize}_{\ext{\varphi}{y}{y}}(N)
\end{align*}

\subsection{Delay (Clos3 $\to$ Clos4)}

\begin{align*}
  \pass{delay}(\clos{r_1\ldots r_n}{e\,y} N) &=
    \pair{\fun{e\,y.\,\pass{delay}(N)}}{\tuple{r_1,\ldots,r_n}} \\
  \pass{delay}(L\;M) &=
    \Let f = \pass{delay}(L) \In (\FST f)(\SND f, \pass{delay}(M))
    \qquad f \text{ fresh}
\end{align*}

\subsection{Globalize (Clos4 $\to$ Clos5)}

\[
  \vec{d} = d_0, \ldots, d_{m-1}
  \qquad
  \pass{glob}(M, \vec{d}) = (M', \vec{d}')
\]
\begin{align*}
  \pass{glob}(x, \vec{d}) &= (x, \vec{d}) \\
  \pass{glob}(\fun{e\,y.\,N}, \vec{d}) &= (\funref{|\vec{d}_1|},\; \vec{d}_1, (e\,y.\,N'))
     \quad\text{where } (N', \vec{d}_1) = \pass{glob}(N, \vec{d}) \\
  \pass{glob}(\mathit{op}(\bar{x}_1.A_1, \ldots, \bar{x}_n.A_n), \vec{d}_0)
     &= (\mathit{op}(\bar{x}_1.A'_1, \ldots, \bar{x}_n.A'_n), \vec{d}_n)
     \quad\text{where } (A'_i, \vec{d}_i) = \pass{glob}(A_i, \vec{d}_{i-1}) \\[1ex]
  \pass{globalize}(M) &= (\vec{d};\; M') \quad\text{where } (M', \vec{d}) = \pass{glob}(M, \epsilon)
\end{align*}

\section{Relations}

\subsection{Annotate and Enclose}

\begin{align*}
  \mathit{Enclosed}(M) &\iff \text{every closure } \clos{x_1\ldots x_n}{y} N \text{ in } M \text{ has }
    \FV(\clos{x_1\ldots x_n}{y} N) = \{x_1,\ldots,x_n\}
\end{align*}

\begin{lemma}[Correctness of Annotate]\label{lem:annotate}
If $\rho_1(x) \simeq \rho_2(x)$ for every $x \in \FV(M)$
and $\rho_2(x) \NE$ for every $x$, then
\[
  \SEM{M}^{\mathrm{ISWIM}}\rho_1 \simeq \SEM{\pass{annotate}(M)}^{\mathrm{1}}\rho_2 .
\]
\end{lemma}

\begin{lemma}[Correctness of Enclose]\label{lem:enclose}
If $\mathit{Enclosed}(M)$ and $\rho_1(x) \simeq \rho_2(x)$ for every $x \in \FV(M)$, then
\[

\subsection{Delay}

\paragraph{Depth.}
\begin{align*}
  \mathit{depth}(c) = \mathit{depth}(\nu) = \mathit{depth}(\omega) = \mathit{depth}(\emptytup) &= 0 \\
  \mathit{depth}(V \mapsto w) &= 1 + \max(\mathit{depth}(V), \mathit{depth}(w)) \\
  \mathit{depth}(\carv{u}) &= 1 + \mathit{depth}(u) \\
  \mathit{depth}(\cdrv{v}) &= 1 + \mathit{depth}(v) \\
  \mathit{depth}(\tup{i}{d}) = \mathit{depth}(\Left{d}) = \mathit{depth}(\Right{d})
    &= 1 + \mathit{depth}(d) \\
  \mathit{depth}(V) &= \max_{v \in V} \mathit{depth}(v) \\[1ex]
  \Bnd{k}{V} &\iff \forall v \in V.\; \mathit{depth}(v) < k
\end{align*}

\paragraph{The logical relation.}
Parameterized by
\begin{align*}
\mathit{Entry}& : \FSET{\V} \to \FSET{\V} \to \V \to \mathsf{Prop}, \\
\circledast & : \SET{\V} \to \SET{\V} \to \SET{\V}, \text{ and } \\
\mathit{Good} &: \FSET{\V} \to \mathsf{Prop}.
\end{align*}

\begin{align*}
  \mathit{Calls}(V) &= \{ (U_1 \cup \cdots \cup U_n,\, \{w_1, \ldots, w_n\}) \mid
     \mathit{Entry}(V, U_i, w_i),\; U_i \NE \text{ for } 1 \leq i \leq n \} \\[1ex]
  \Rel{0}{V}{D} &\iff \mathsf{true} \\
  \Rel{k+1}{V}{D} &\iff
  \begin{array}[t]{l}
    \forall c \in V.\; c \in D \\
    \land\; (\omega \in V \implies \omega \in D) \\
    \land\; (V \NE \implies D \NE) \\
    \land\; \forall (U, W) \in \mathit{Calls}(V).\; \forall E.\;
       \mathit{Good}(U) \land \Rel{k}{U}{E} \implies \Rel{k}{W}{D \circledast E} \\
    \land\; \forall i\, W.\; W \subseteq \{ d \mid \tup{i}{d} \in V \}
       \implies \Rel{k}{W}{\key{proj}_i(D)} \\
    \land\; \forall W.\; W \subseteq \{ d \mid \Left{d} \in V \}
       \implies \Rel{k}{W}{\{ d \mid \Left{d} \in D \}} \\
    \land\; \forall W.\; W \subseteq \{ d \mid \Right{d} \in V \}
       \implies \Rel{k}{W}{\{ d \mid \Right{d} \in D \}}
  \end{array} \\[1ex]
  \rho_1 \rhd \rho_2 &\iff \forall x\, V\, k.\; V \subseteq \rho_1(x) \land \Bnd{k}{V}
     \implies \Rel{k}{V}{\rho_2(x)}
\end{align*}

\paragraph{Preservation instance} (observations of Clos3, sets of Clos4).
\begin{align*}
  \mathit{Entry}(V, U_0, w) &\iff (U_0 \mapsto w) \in V \\
  D \circledast E &= D \odot E \\
  \mathit{Good}(U) &= \mathsf{true}
\end{align*}

\paragraph{Reflection instance} (observations of Clos4, sets of Clos3).
\begin{align*}
  \mathit{Entry}(V', U_0, w) &\iff \exists F \NE.\;
     (F \mapsto (U_0 \mapsto w)) \in \key{car}(V'),\; F \subseteq \key{cdr}(V') \\
  D \circledast E &= D \app E \\
  \mathit{Good}(U') &= \consis(U')
\end{align*}

\begin{lemma}[Continuity]\label{lem:continuity}
$\SEM{M}^{\mathrm{3}}$ and $\SEM{M}^{\mathrm{4}}$ are continuous.
\end{lemma}

\begin{lemma}[Consistency of Clos4]\label{lem:consis4}

\end{lemma}

\begin{lemma}[Delayed Application]\label{lem:let-app}

  \SEM{\Let f = L \In (\FST f)(\SND f, N)}^{\mathrm{4}}\rho
  \simeq \SEM{L}^{\mathrm{4}}\rho \odot \SEM{N}^{\mathrm{4}}\rho .
\]
\end{lemma}

\begin{lemma}[Delay Preserves]\label{lem:delay-preserve}
In the preservation instance, if $\rho'(x) \NE$ for every $x$ and
$\rho \rhd \rho'$, then for every $k$ and every $V$ with $\Bnd{k}{V}$,
\[
  V \subseteq \SEM{M}^{\mathrm{3}}\rho
  \implies \Rel{k}{V}{\SEM{\pass{delay}(M)}^{\mathrm{4}}\rho'} .

  \implies \Rel{k}{V'}{\SEM{M}^{\mathrm{3}}\rho} .

\end{lemma}

\subsection{Globalize}

\begin{align*}
  \mathit{Extends}(\vec{d}, \vec{d}') &\iff \vec{d} \text{ is a prefix of } \vec{d}' \\

  \mathit{Agree}_n(T, T') &\iff \forall k < n.\; T(k) \simeq T'(k)

\begin{lemma}[Well-Formed Tables]\label{lem:glob-wf}

$\SEM{M}^{\mathrm{5}}_T\rho \simeq \SEM{M}^{\mathrm{5}}_{T'}\rho'$.
\end{lemma}

\begin{lemma}[Correctness of Glob]\label{lem:glob}

$\SEM{M}^{\mathrm{4}}\rho \simeq \SEM{M'}^{\mathrm{5}}_{\key{table}(\vec{d}')}\rho'$.
\end{lemma}
